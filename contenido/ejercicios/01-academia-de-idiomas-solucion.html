2 y 7. Academia, que es el lado «uno» de la relación bidireccional con Profesor. Lleva mappedBy y no lleva cascada: así, borrar una academia no borra a sus profesores.

package com.academia.babel.entity;

import java.util.ArrayList;
import java.util.List;

import jakarta.persistence.Column;
import jakarta.persistence.Entity;
import jakarta.persistence.GeneratedValue;
import jakarta.persistence.GenerationType;
import jakarta.persistence.Id;
import jakarta.persistence.OneToMany;
import jakarta.persistence.Table;

@Entity
@Table(name = "academias")
public class Academia {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    @Column(name = "id_academia")
    private Long id;

    @Column(nullable = false, length = 80)
    private String nombre;

    @Column(nullable = false)
    private String ciudad;

    // Lado inverso: la clave foránea está en Profesor. Sin cascade: borrar la academia no borra a los profesores.
    @OneToMany(mappedBy = "academia")
    private List<Profesor> profesores = new ArrayList<>();

    public Academia() {}

    public Academia(String nombre, String ciudad) {
        this.nombre = nombre;
        this.ciudad = ciudad;
    }

    // Mantiene sincronizados los dos lados de la relación
    public void addProfesor(Profesor profesor) {
        profesores.add(profesor);
        profesor.setAcademia(this);
    }

    public Long getId() { return id; }
    public String getNombre() { return nombre; }
    public String getCiudad() { return ciudad; }
    public List<Profesor> getProfesores() { return profesores; }
}

4, 8 y 9. Alumno. Es el propietario del 1:1 con Expediente (id_expediente, obligatoria y única), con cascada total. Y es el lado «uno» de Matricula, con orphanRemoval. Fíjate en el constructor: recibe el expediente, así que es imposible crear un alumno sin él.

package com.academia.babel.entity;

import java.time.LocalDate;
import java.util.ArrayList;
import java.util.List;

import jakarta.persistence.CascadeType;
import jakarta.persistence.Column;
import jakarta.persistence.Entity;
import jakarta.persistence.GeneratedValue;
import jakarta.persistence.GenerationType;
import jakarta.persistence.Id;
import jakarta.persistence.JoinColumn;
import jakarta.persistence.OneToMany;
import jakarta.persistence.OneToOne;
import jakarta.persistence.Table;

@Entity
@Table(name = "alumnos")
public class Alumno {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    @Column(name = "id_alumno")
    private Long id;

    @Column(nullable = false)
    private String nombre;

    @Column(nullable = false, unique = true)
    private String email;

    @Column(name = "fecha_nacimiento", nullable = false)
    private LocalDate fechaNacimiento;

    // Propietario del 1:1. La cascada guarda y borra el expediente junto con el alumno.
    @OneToOne(cascade = CascadeType.ALL, optional = false)
    @JoinColumn(name = "id_expediente", nullable = false, unique = true)
    private Expediente expediente;

    // Lado inverso del N:M con datos: las matrículas dependen del alumno
    @OneToMany(mappedBy = "alumno", cascade = CascadeType.ALL, orphanRemoval = true)
    private List<Matricula> matriculas = new ArrayList<>();

    // Lo necesita JPA para crear objetos al leer de la base. No lo uses tú.
    public Alumno() {}

    // Única forma de crear un alumno: obliga a darle su expediente
    public Alumno(String nombre, String email, LocalDate fechaNacimiento, Expediente expediente) {
        this.nombre = nombre;
        this.email = email;
        this.fechaNacimiento = fechaNacimiento;
        this.expediente = expediente;
    }

    public void addMatricula(Matricula matricula) {
        matriculas.add(matricula);
        matricula.setAlumno(this);
    }

    public Long getId() { return id; }
    public String getNombre() { return nombre; }
    public String getEmail() { return email; }
    public LocalDate getFechaNacimiento() { return fechaNacimiento; }
    public Expediente getExpediente() { return expediente; }
    public List<Matricula> getMatriculas() { return matriculas; }
}

9. Matricula es la entidad que representa la relación N:M con datos. Tiene dos @ManyToOne (propietaria de ambas claves foráneas). Fíjate en que nota es un Double y no un double, porque tiene que poder valer null.

package com.academia.babel.entity;

import java.time.LocalDate;

import com.academia.babel.enums.EstadoMatricula;

import jakarta.persistence.Column;
import jakarta.persistence.Entity;
import jakarta.persistence.EnumType;
import jakarta.persistence.Enumerated;
import jakarta.persistence.FetchType;
import jakarta.persistence.GeneratedValue;
import jakarta.persistence.GenerationType;
import jakarta.persistence.Id;
import jakarta.persistence.JoinColumn;
import jakarta.persistence.ManyToOne;
import jakarta.persistence.Table;

@Entity
@Table(name = "matriculas")
public class Matricula {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    @Column(name = "id_matricula")
    private Long id;

    @ManyToOne(fetch = FetchType.LAZY, optional = false)
    @JoinColumn(name = "id_alumno", nullable = false)
    private Alumno alumno;

    @ManyToOne(fetch = FetchType.LAZY, optional = false)
    @JoinColumn(name = "id_curso", nullable = false)
    private Curso curso;

    @Column(nullable = false)
    private LocalDate fecha;

    // Double (con mayúscula) porque puede ser nula: al matricularse todavía no hay nota
    private Double nota;

    @Enumerated(EnumType.STRING)
    @Column(nullable = false, length = 20)
    private EstadoMatricula estado;

    private boolean pagada;

    public Matricula() {}

    // El alumno no va aquí: se asigna con alumno.addMatricula(...)
    public Matricula(Curso curso, LocalDate fecha, EstadoMatricula estado, boolean pagada) {
        this.curso = curso;
        this.fecha = fecha;
        this.estado = estado;
        this.pagada = pagada;
    }

    public Long getId() { return id; }
    public Alumno getAlumno() { return alumno; }
    public void setAlumno(Alumno alumno) { this.alumno = alumno; }
    public Curso getCurso() { return curso; }
    public LocalDate getFecha() { return fecha; }
    public Double getNota() { return nota; }
    public EstadoMatricula getEstado() { return estado; }
    public boolean isPagada() { return pagada; }
}

10. Respuestas:

  • a) Porque la relación tiene datos propios (fecha, nota, estado y si está pagada). Un @ManyToMany directo solo crea una tabla intermedia con dos ids y no puede guardar nada más. Con Matricula la relación pasa a ser dos @ManyToOne y la tabla intermedia se convierte en una entidad con sus propias columnas.
  • b) Las claves foráneas: profesores.id_academia, alumnos.id_expediente, matriculas.id_alumno y matriculas.id_curso. Siempre está en el lado propietario (el que lleva @JoinColumn); el lado con mappedBy no tiene columna.
  • c) Con EAGER, al cargar 100 matrículas Hibernate carga también el alumno y el curso de cada una, aunque no los necesites. Pueden dispararse muchas consultas de más. Con LAZY, solo se cargan cuando se piden.

14. Alumnos. La consulta de «matriculados en un curso» necesita JOIN (hay que pasar por matriculas) y DISTINCT (un alumno podría aparecer repetido). El filtro por nivel navega alumno → expediente → nivelInicial, y Spring lo entiende solo por el nombre.

package com.academia.babel.repository;

17. La clase principal, con el run() que crea un alumno nuevo y lo matricula:

package com.academia.babel;

import java.time.LocalDate;

import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.CommandLineRunner;
import org.springframework.boot.SpringApplication;
import org.springframework.boot.autoconfigure.SpringBootApplication;

import com.academia.babel.entity.Alumno;
import com.academia.babel.entity.Curso;
import com.academia.babel.entity.Expediente;
import com.academia.babel.entity.Matricula;
import com.academia.babel.enums.EstadoMatricula;
import com.academia.babel.enums.Nivel;
import com.academia.babel.repository.IAlumnoRepository;
import com.academia.babel.repository.ICursoRepository;

@SpringBootApplication
public class BabelApplication implements CommandLineRunner {

    @Autowired
    private IAlumnoRepository alumnoRepository;

    @Autowired
    private ICursoRepository cursoRepository;

    public static void main(String[] args) {
        SpringApplication.run(BabelApplication.class, args);
    }

    @Override
    public void run(String... args) {
        // 1) El curso ya existe (lo cargó data.sql): solo hay que buscarlo
        Curso curso = cursoRepository.findById(1L).orElseThrow();

        // 2) El expediente se crea primero porque el constructor de Alumno lo exige
        Expediente expediente = new Expediente("AL-0004", Nivel.A2, null);
        Alumno alumno = new Alumno("Pablo León", "pablo@mail.com", LocalDate.of(2001, 3, 9), expediente);

        // 3) La matrícula se cuelga del alumno
        alumno.addMatricula(new Matricula(curso, LocalDate.now(), EstadoMatricula.ACTIVA, false));

        // 4) Un solo save: la cascada guarda el expediente y la matrícula
        alumnoRepository.save(alumno);
        System.out.println("Alumno guardado con id " + alumno.getId());
    }
}

Por qué este orden: el curso tiene que existir antes, porque la matrícula apunta a él. El expediente se crea antes que el alumno porque el constructor lo exige. Y gracias a cascade = ALL, un solo save(alumno) inserta el expediente (primero, porque el alumno tiene su clave foránea), el alumno y, por último, la matrícula, que necesita el id del alumno.

Y la implementación. Cada método que lee relaciones perezosas va con @Transactional(readOnly = true), y la conversión a DTO se hace dentro del servicio:

package com.academia.babel.service.impl;

import java.util.ArrayList;
import java.util.List;
import java.util.Optional;

import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import com.academia.babel.dto.AlumnoRequestDTO;
import com.academia.babel.dto.AlumnoResponseDTO;
import com.academia.babel.dto.MatriculaRequestDTO;
import com.academia.babel.dto.MatriculaResumenDTO;
import com.academia.babel.entity.Alumno;
import com.academia.babel.entity.Curso;
import com.academia.babel.entity.Expediente;
import com.academia.babel.entity.Matricula;
import com.academia.babel.enums.EstadoMatricula;
import com.academia.babel.enums.Nivel;
import com.academia.babel.repository.IAlumnoRepository;
import com.academia.babel.repository.ICursoRepository;
import com.academia.babel.repository.IMatriculaRepository;
import com.academia.babel.service.IAlumnoService;

@Service
public class AlumnoServiceImpl implements IAlumnoService {

    @Autowired
    private IAlumnoRepository alumnoRepository;

    @Autowired
    private ICursoRepository cursoRepository;

    @Autowired
    private IMatriculaRepository matriculaRepository;

    @Override
    public boolean existeEmailOMatricula(AlumnoRequestDTO dto) {
        return alumnoRepository.existsByEmail(dto.email())
                || alumnoRepository.existsByExpedienteNumeroMatricula(dto.numeroMatricula());
    }

    @Override
    @Transactional
    public AlumnoResponseDTO crear(AlumnoRequestDTO dto) {
        // DTO -> Entity: el expediente se crea primero porque el constructor de Alumno lo exige
        Expediente expediente = new Expediente(dto.numeroMatricula(), dto.nivelInicial(), null);
        Alumno alumno = new Alumno(dto.nombre(), dto.email(), dto.fechaNacimiento(), expediente);
        return mapearADTO(alumnoRepository.save(alumno));
    }

    @Override
    @Transactional(readOnly = true)
    public AlumnoResponseDTO obtenerPorId(Long id) {
        Optional<Alumno> alumnoOpt = alumnoRepository.findById(id);
        if (alumnoOpt.isEmpty()) {
            return null;   // el alumno no existe
        }
        return mapearADTO(alumnoOpt.get());
    }

    @Override
    @Transactional(readOnly = true)
    public List<AlumnoResponseDTO> listarPorNivel(Nivel nivel) {
        List<AlumnoResponseDTO> lista = new ArrayList<>();
        for (Alumno a : alumnoRepository.findByExpedienteNivelInicial(nivel)) {
            lista.add(mapearADTO(a));
        }
        return lista;
    }

    @Override
    public boolean tieneMatriculaActiva(Long alumnoId, Long cursoId) {
        return matriculaRepository.existsByAlumnoIdAndCursoIdAndEstado(alumnoId, cursoId, EstadoMatricula.ACTIVA);
    }

    @Override
    @Transactional
    public AlumnoResponseDTO matricular(Long alumnoId, MatriculaRequestDTO dto) {
        Optional<Alumno> alumnoOpt = alumnoRepository.findById(alumnoId);
        Optional<Curso> cursoOpt = cursoRepository.findById(dto.cursoId());
        if (alumnoOpt.isEmpty() || cursoOpt.isEmpty()) {
            return null;   // falta el alumno o el curso
        }
        Alumno alumno = alumnoOpt.get();
        Matricula matricula = new Matricula(cursoOpt.get(), dto.fecha(), EstadoMatricula.ACTIVA, false);
        alumno.addMatricula(matricula);
        matriculaRepository.save(matricula);
        return mapearADTO(alumno);
    }

    @Override
    public boolean tieneMatriculasActivas(Long id) {
        return matriculaRepository.countByAlumnoIdAndEstado(id, EstadoMatricula.ACTIVA) > 0;
    }

    @Override
    public void eliminar(Long id) {
        alumnoRepository.deleteById(id);   // la cascada borra su expediente y sus matrículas
    }

    // Entity -> DTO (recorre las matrículas, que son LAZY: por eso se hace dentro de la transacción)
    private AlumnoResponseDTO mapearADTO(Alumno a) {
        List<MatriculaResumenDTO> matriculas = new ArrayList<>();
        for (Matricula m : a.getMatriculas()) {
            matriculas.add(new MatriculaResumenDTO(m.getCurso().getTitulo(), m.getFecha(), m.getEstado(), m.getNota()));
        }
        return new AlumnoResponseDTO(a.getId(), a.getNombre(), a.getEmail(),
                a.getExpediente().getNumeroMatricula(), a.getExpediente().getNivelInicial(), matriculas);
    }
}

Cómo comunica cada resultado: el alumno o el curso que no existen devuelven null, que el controlador convierte en 404. Los conflictos (email o número de matrícula repetidos, matrícula activa repetida, alumno con matrículas activas) se preguntan antes con un método boolean, y el controlador responde 409. Una lista vacía no es error: se devuelve vacía (200).

@RestController

        }

        URI location = ServletUriComponentsBuilder.fromCurrentRequest()
                .path("/{id}").buildAndExpand(creado.id()).toUri();
        return ResponseEntity.created(location).body(creado);
    }

    @GetMapping("/{id}")
    public ResponseEntity<AlumnoResponseDTO> obtener(@PathVariable Long id) {
        AlumnoResponseDTO alumno = alumnoService.obtenerPorId(id);
        if (alumno == null) {

        }

    }

    // POST /api/alumnos/{id}/matriculas -> 201 | 400 | 404 (alumno o curso) | 409 (ya matriculado)
    @PostMapping("/{id}/matriculas")
    public ResponseEntity<AlumnoResponseDTO> matricular(@PathVariable Long id,
            @Valid @RequestBody MatriculaRequestDTO dto) {
        if (alumnoService.tieneMatriculaActiva(id, dto.cursoId())) {
            return ResponseEntity.status(HttpStatus.CONFLICT).build();   // 409
        }
        AlumnoResponseDTO alumno = alumnoService.matricular(id, dto);
        if (alumno == null) {
            return ResponseEntity.notFound().build();                    // 404
        }
        return ResponseEntity.status(HttpStatus.CREATED).body(alumno);
    }

    // DELETE /api/alumnos/{id} -> 204 | 404 | 409 (matrículas activas)
    @DeleteMapping("/{id}")
    public ResponseEntity<Void> eliminar(@PathVariable Long id) {
        if (alumnoService.obtenerPorId(id) == null) {
            return ResponseEntity.notFound().build();                    // 404
        }
        if (alumnoService.tieneMatriculasActivas(id)) {
            return ResponseEntity.status(HttpStatus.CONFLICT).build();   // 409
        }
        alumnoService.eliminar(id);
        return ResponseEntity.noContent().build();                       // 204
    }
}

Parte 5: preguntas de comprensión

22. LazyInitializationException. Ocurre al leer una relación LAZY (aquí, las matrículas del alumno o el curso de cada matrícula) cuando la sesión de Hibernate ya se ha cerrado: por ejemplo, si el servicio devuelve la entidad y es el controlador quien intenta recorrerla. Se evita convirtiendo a DTO dentro del servicio y anotándolo con @Transactional (readOnly = true si solo lee), para que la sesión siga abierta mientras se recorren las relaciones.

23. El problema N+1. Una consulta trae los 50 alumnos y después, al recorrer las matrículas de cada uno, Hibernate lanza otra consulta por alumno (y otra más por cada curso): 1 + N. Dos soluciones: un JOIN FETCH en la consulta, o @EntityGraph sobre el método del repositorio. Por ejemplo:

@Query("SELECT DISTINCT a FROM Alumno a "
     + "LEFT JOIN FETCH a.matriculas m LEFT JOIN FETCH m.curso "
     + "JOIN FETCH a.expediente")
List<Alumno> listarConMatriculas();

@EntityGraph(attributePaths = {"expediente", "matriculas", "matriculas.curso"})
List<Alumno> findByExpedienteNivelInicial(Nivel nivel);

24. El constructor de Alumno exige el expediente. La relación es obligatoria: un alumno sin expediente no puede guardarse. Si se asignara después con un setter, nada impediría olvidarlo, y el fallo solo aparecería al ejecutar y guardar (con un error de base de datos). Con el constructor, el compilador no deja crear un alumno incompleto: el error sale antes, y en rojo, en el editor. El constructor vacío existe solo porque JPA lo necesita al leer filas.

25. Borrar una academia con profesores. El 500 viene de la base de datos (violación de clave foránea: los profesores siguen apuntando a ella). Debería devolver 409 Conflict. Falta la comprobación previa, igual que con las matrículas activas: el servicio la ofrece y el controlador responde 409 si hay profesores:

public boolean tieneProfesores(Long id) {
    return profesorRepository.countByAcademiaId(id) > 0;
}

26. El esquema de tablas. Las claves foráneas (FK) están siempre en el lado propietario:

academias    (id_academia PK, nombre, ciudad)
profesores   (id_profesor PK, nombre, salario_hora, idioma, activo,
              id_academia FK -> academias)                    1:N
expedientes  (id_expediente PK, numero_matricula UNIQUE, nivel_inicial, observaciones)
alumnos      (id_alumno PK, nombre, email UNIQUE, fecha_nacimiento,
              id_expediente FK UNIQUE -> expedientes)          1:1
cursos       (id_curso PK, titulo, idioma, nivel, precio)
matriculas   (id_matricula PK, fecha, nota, estado, pagada,
              id_alumno FK -> alumnos,                         N:M con datos:
              id_curso  FK -> cursos)                          dos 1:N hacia Matricula

Parte 4 (3 puntos): los DTO con sus validaciones, la conversión solo en el servicio, el controlador sin lógica de negocio y los códigos 201, 200, 204, 400, 404 y 409 bien elegidos.
