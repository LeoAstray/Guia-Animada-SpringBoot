2 y 7. Academia, que es el lado «uno» de la relación bidireccional con Profesor. Lleva mappedBy y no lleva cascada: así, borrar una academia no borra a sus profesores.

package com.academia.babel.entity;

import java.util.ArrayList;
import java.util.List;

import jakarta.persistence.Column;
import jakarta.persistence.Entity;
import jakarta.persistence.GeneratedValue;
import jakarta.persistence.GenerationType;
import jakarta.persistence.Id;
import jakarta.persistence.OneToMany;
import jakarta.persistence.Table;

@Entity
@Table(name = "academias")
public class Academia {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    @Column(name = "id_academia")
    private Long id;

    @Column(nullable = false, length = 80)
    private String nombre;

    @Column(nullable = false)
    private String ciudad;

    // Lado inverso: la clave foránea está en Profesor. Sin cascade: borrar la academia no borra a los profesores.
    @OneToMany(mappedBy = "academia")
    private List<Profesor> profesores = new ArrayList<>();

    public Academia() {}

    public Academia(String nombre, String ciudad) {
        this.nombre = nombre;
        this.ciudad = ciudad;
    }

    // Mantiene sincronizados los dos lados de la relación
    public void addProfesor(Profesor profesor) {
        profesores.add(profesor);
        profesor.setAcademia(this);
    }

    public Long getId() { return id; }
    public String getNombre() { return nombre; }
    public String getCiudad() { return ciudad; }
    public List<Profesor> getProfesores() { return profesores; }
}

4, 8 y 9. Alumno. Es el propietario del 1:1 con Expediente (id_expediente, obligatoria y única), con cascada total. Y es el lado «uno» de Matricula, con orphanRemoval. Fíjate en el constructor: recibe el expediente, así que es imposible crear un alumno sin él.

package com.academia.babel.entity;

import java.time.LocalDate;
import java.util.ArrayList;
import java.util.List;

import jakarta.persistence.CascadeType;
import jakarta.persistence.Column;
import jakarta.persistence.Entity;
import jakarta.persistence.GeneratedValue;
import jakarta.persistence.GenerationType;
import jakarta.persistence.Id;
import jakarta.persistence.JoinColumn;
import jakarta.persistence.OneToMany;
import jakarta.persistence.OneToOne;
import jakarta.persistence.Table;

@Entity
@Table(name = "alumnos")
public class Alumno {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    @Column(name = "id_alumno")
    private Long id;

    @Column(nullable = false)
    private String nombre;

    @Column(nullable = false, unique = true)
    private String email;

    @Column(name = "fecha_nacimiento", nullable = false)
    private LocalDate fechaNacimiento;

    // Propietario del 1:1. La cascada guarda y borra el expediente junto con el alumno.
    @OneToOne(cascade = CascadeType.ALL, optional = false)
    @JoinColumn(name = "id_expediente", nullable = false, unique = true)
    private Expediente expediente;

    // Lado inverso del N:M con datos: las matrículas dependen del alumno
    @OneToMany(mappedBy = "alumno", cascade = CascadeType.ALL, orphanRemoval = true)
    private List<Matricula> matriculas = new ArrayList<>();

    // Lo necesita JPA para crear objetos al leer de la base. No lo uses tú.
    public Alumno() {}

    // Única forma de crear un alumno: obliga a darle su expediente
    public Alumno(String nombre, String email, LocalDate fechaNacimiento, Expediente expediente) {
        this.nombre = nombre;
        this.email = email;
        this.fechaNacimiento = fechaNacimiento;
        this.expediente = expediente;
    }

    public void addMatricula(Matricula matricula) {
        matriculas.add(matricula);
        matricula.setAlumno(this);
    }

    public Long getId() { return id; }
    public String getNombre() { return nombre; }
    public String getEmail() { return email; }
    public LocalDate getFechaNacimiento() { return fechaNacimiento; }
    public Expediente getExpediente() { return expediente; }
    public List<Matricula> getMatriculas() { return matriculas; }
}

9. Matricula es la entidad que representa la relación N:M con datos. Tiene dos @ManyToOne (propietaria de ambas claves foráneas). Fíjate en que nota es un Double y no un double, porque tiene que poder valer null.

package com.academia.babel.entity;

import java.time.LocalDate;

import com.academia.babel.enums.EstadoMatricula;

import jakarta.persistence.Column;
import jakarta.persistence.Entity;
import jakarta.persistence.EnumType;
import jakarta.persistence.Enumerated;
import jakarta.persistence.FetchType;
import jakarta.persistence.GeneratedValue;
import jakarta.persistence.GenerationType;
import jakarta.persistence.Id;
import jakarta.persistence.JoinColumn;
import jakarta.persistence.ManyToOne;
import jakarta.persistence.Table;

@Entity
@Table(name = "matriculas")
public class Matricula {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    @Column(name = "id_matricula")
    private Long id;

    @ManyToOne(fetch = FetchType.LAZY, optional = false)
    @JoinColumn(name = "id_alumno", nullable = false)
    private Alumno alumno;

    @ManyToOne(fetch = FetchType.LAZY, optional = false)
    @JoinColumn(name = "id_curso", nullable = false)
    private Curso curso;

    @Column(nullable = false)
    private LocalDate fecha;

    // Double (con mayúscula) porque puede ser nula: al matricularse todavía no hay nota
    private Double nota;

    @Enumerated(EnumType.STRING)
    @Column(nullable = false, length = 20)
    private EstadoMatricula estado;

    private boolean pagada;

    public Matricula() {}

    // El alumno no va aquí: se asigna con alumno.addMatricula(...)
    public Matricula(Curso curso, LocalDate fecha, EstadoMatricula estado, boolean pagada) {
        this.curso = curso;
        this.fecha = fecha;
        this.estado = estado;
        this.pagada = pagada;
    }

    public Long getId() { return id; }
    public Alumno getAlumno() { return alumno; }
    public void setAlumno(Alumno alumno) { this.alumno = alumno; }
    public Curso getCurso() { return curso; }
    public LocalDate getFecha() { return fecha; }
    public Double getNota() { return nota; }

10. Respuestas:

  • a) Porque la relación tiene datos propios (fecha, nota, estado y si está pagada). Un @ManyToMany directo solo crea una tabla intermedia con dos ids y no puede guardar nada más. Con Matricula la relación pasa a ser dos @ManyToOne y la tabla intermedia se convierte en una entidad con sus propias columnas.
  • b) Las claves foráneas: profesores.id_academia, alumnos.id_expediente, matriculas.id_alumno y matriculas.id_curso. Siempre está en el lado propietario (el que lleva @JoinColumn); el lado con mappedBy no tiene columna.
  • c) Con EAGER, al cargar 100 matrículas Hibernate carga también el alumno y el curso de cada una, aunque no los necesites. Pueden dispararse muchas consultas de más. Con LAZY, solo se cargan cuando se piden.

package com.academia.babel.service.impl;

import java.util.ArrayList;
import java.util.List;

import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import com.academia.babel.dto.AlumnoRequestDTO;
import com.academia.babel.dto.AlumnoResponseDTO;
import com.academia.babel.dto.MatriculaRequestDTO;
import com.academia.babel.dto.MatriculaResumenDTO;
import com.academia.babel.entity.Alumno;
import com.academia.babel.entity.Curso;
import com.academia.babel.entity.Expediente;
import com.academia.babel.entity.Matricula;
import com.academia.babel.enums.EstadoMatricula;
import com.academia.babel.enums.Nivel;
import com.academia.babel.exception.ConflictoException;
import com.academia.babel.exception.NoEncontradoException;
import com.academia.babel.repository.IAlumnoRepository;
import com.academia.babel.repository.ICursoRepository;
import com.academia.babel.repository.IMatriculaRepository;
import com.academia.babel.service.IAlumnoService;

@Service
public class AlumnoServiceImpl implements IAlumnoService {

    @Autowired
    private IAlumnoRepository alumnoRepository;

    @Autowired
    private ICursoRepository cursoRepository;

    @Autowired
    private IMatriculaRepository matriculaRepository;

    @Override
    @Transactional
    public AlumnoResponseDTO crear(AlumnoRequestDTO dto) {
        if (alumnoRepository.existsByEmail(dto.email())) {
            throw new ConflictoException("Ya existe un alumno con ese email");
        }
        if (alumnoRepository.existsByExpedienteNumeroMatricula(dto.numeroMatricula())) {
            throw new ConflictoException("Ya existe un alumno con ese número de matrícula");
        }
        // DTO -> Entity: el expediente se crea primero porque el constructor de Alumno lo exige
        Expediente expediente = new Expediente(dto.numeroMatricula(), dto.nivelInicial(), null);
        Alumno alumno = new Alumno(dto.nombre(), dto.email(), dto.fechaNacimiento(), expediente);
        return mapearADTO(alumnoRepository.save(alumno));
    }

    @Override
    @Transactional(readOnly = true)
    public AlumnoResponseDTO obtenerPorId(Long id) {
        return mapearADTO(buscarAlumno(id));
    }

    @Override
    @Transactional(readOnly = true)
    public List<AlumnoResponseDTO> listarPorNivel(Nivel nivel) {
        List<AlumnoResponseDTO> lista = new ArrayList<>();
        for (Alumno a : alumnoRepository.findByExpedienteNivelInicial(nivel)) {
            lista.add(mapearADTO(a));
        }
        return lista;
    }

    @Override
    @Transactional
    public AlumnoResponseDTO matricular(Long alumnoId, MatriculaRequestDTO dto) {
        Alumno alumno = buscarAlumno(alumnoId);
        Curso curso = cursoRepository.findById(dto.cursoId())
                .orElseThrow(() -> new NoEncontradoException("No existe el curso " + dto.cursoId()));
        if (matriculaRepository.existsByAlumnoIdAndCursoIdAndEstado(alumnoId, dto.cursoId(), EstadoMatricula.ACTIVA)) {
            throw new ConflictoException("El alumno ya tiene una matrícula activa en ese curso");
        }
        Matricula matricula = new Matricula(curso, dto.fecha(), EstadoMatricula.ACTIVA, false);
        alumno.addMatricula(matricula);
        matriculaRepository.save(matricula);
        return mapearADTO(alumno);
    }

    @Override
    @Transactional
    public void eliminar(Long id) {
        Alumno alumno = buscarAlumno(id);
        if (matriculaRepository.countByAlumnoIdAndEstado(id, EstadoMatricula.ACTIVA) > 0) {
            throw new ConflictoException("El alumno tiene matrículas activas");
        }
        alumnoRepository.delete(alumno);   // la cascada borra su expediente y sus matrículas
    }

    private Alumno buscarAlumno(Long id) {
        return alumnoRepository.findById(id)
                .orElseThrow(() -> new NoEncontradoException("No existe el alumno " + id));
    }

    // Entity -> DTO (recorre las matrículas, que son LAZY: por eso se hace dentro de la transacción)
    private AlumnoResponseDTO mapearADTO(Alumno a) {
        List<MatriculaResumenDTO> matriculas = new ArrayList<>();
        for (Matricula m : a.getMatriculas()) {
            matriculas.add(new MatriculaResumenDTO(m.getCurso().getTitulo(), m.getFecha(), m.getEstado(), m.getNota()));
        }
        return new AlumnoResponseDTO(a.getId(), a.getNombre(), a.getEmail(),
                a.getExpediente().getNumeroMatricula(), a.getExpediente().getNivelInicial(), matriculas);
    }
}

Parte 5: preguntas de comprensión

22. LazyInitializationException. Ocurre al leer una relación LAZY (aquí, las matrículas del alumno o el curso de cada matrícula) cuando la sesión de Hibernate ya se ha cerrado: por ejemplo, si el servicio devuelve la entidad y es el controlador quien intenta recorrerla. Se evita convirtiendo a DTO dentro del servicio y anotándolo con @Transactional (readOnly = true si solo lee), para que la sesión siga abierta mientras se recorren las relaciones.

23. El problema N+1. Una consulta trae los 50 alumnos y después, al recorrer las matrículas de cada uno, Hibernate lanza otra consulta por alumno (y otra más por cada curso): 1 + N. Dos soluciones: un JOIN FETCH en la consulta, o @EntityGraph sobre el método del repositorio. Por ejemplo:

@Query("SELECT DISTINCT a FROM Alumno a "
     + "LEFT JOIN FETCH a.matriculas m LEFT JOIN FETCH m.curso "
     + "JOIN FETCH a.expediente")
List<Alumno> listarConMatriculas();

@EntityGraph(attributePaths = {"expediente", "matriculas", "matriculas.curso"})
List<Alumno> findByExpedienteNivelInicial(Nivel nivel);

24. El constructor de Alumno exige el expediente. La relación es obligatoria: un alumno sin expediente no puede guardarse. Si se asignara después con un setter, nada impediría olvidarlo, y el fallo solo aparecería al ejecutar y guardar (con un error de base de datos). Con el constructor, el compilador no deja crear un alumno incompleto: el error sale antes, y en rojo, en el editor. El constructor vacío existe solo porque JPA lo necesita al leer filas.

7 (pregunta). optional = false es una regla del modelo Java: dice que la relación es obligatoria, así que JPA no permite guardar un paciente sin historial (y puede optimizar sabiendo que nunca es nulo). unique = true (en el @JoinColumn) es una regla de la base de datos: crea una restricción de unicidad en id_historial, de modo que dos pacientes no pueden compartir el mismo historial. Sin ella la relación sería en realidad de uno a muchos. Junto con nullable = false (la columna no admite NULL), dejan la relación como «uno a uno obligatoria» tanto en Java como en MySQL.

9. Respuestas:

  • a) Porque la relación tiene datos propios (fecha, coste, tipo y seguimiento). Un @ManyToMany directo solo crea una tabla intermedia con dos ids y no puede guardar nada más. Con Consulta, la tabla intermedia pasa a ser una entidad con sus propias columnas, y la relación se convierte en dos @ManyToOne.
  • b) Las claves foráneas son medicos.id_hospital, pacientes.id_historial, consultas.id_medico y consultas.id_paciente. Siempre están en el lado propietario (el que lleva @JoinColumn); el lado con mappedBy no tiene columna.
  • c) Al borrar un hospital se borrarían también todos sus médicos (y, por la cadena de relaciones, podría fallar o arrastrar más datos), aunque tengan consultas históricas. Es una pérdida masiva de datos en vez de una protección de integridad. Por eso Hospital no lleva cascada, y borrar un hospital con médicos se responde con un 409.

17. Dos motivos para no devolver las entidades:

  • Recursión infinita. Hospital tiene una lista de Medico y cada Medico apunta a su Hospital: al serializar a JSON, Jackson entra en un bucle (hospital → médicos → hospital → …) y falla con un StackOverflowError o similar. Los DTO, con sus «resúmenes», cortan el bucle.
  • Se expone la base de datos. La entidad lleva campos que el cliente no debe ver o decidir (ids internos, la lista completa de consultas, etc.), y atar la API a la entidad hace que cualquier cambio en el modelo rompa a los clientes. Además, las relaciones LAZY se intentarían leer fuera de la transacción (LazyInitializationException).

import com.hospital.examen.exception.NoEncontradoException;

import com.hospital.examen.repository.IHospitalRepository;

@Service
public class MedicoService implements IMedicoService {

    @Autowired
    private IMedicoRepository medicoRepository;

    @Autowired
    private IHospitalRepository hospitalRepository;

    @Autowired
    private IConsultaRepository consultaRepository;

    @Override
    @Transactional
    public MedicoResponseDTO crear(MedicoRequestDTO dto) {
        Hospital hospital = buscarHospital(dto.hospitalId());
        // DTO -> Entity. Un médico nuevo empieza activo.
        Medico medico = new Medico(dto.nombre(), dto.salario(), dto.especialidad(), true);
        hospital.addMedico(medico);
        return mapearADTO(medicoRepository.save(medico));
    }

    @Override
    @Transactional(readOnly = true)
    public MedicoResponseDTO obtener(Long id) {
        return mapearADTO(buscarMedico(id));
    }

    @Override
    @Transactional(readOnly = true)
    public List<MedicoResponseDTO> listarPorHospital(Long hospitalId) {
        buscarHospital(hospitalId);   // si no existe, lanza "no encontrado"
        List<MedicoResponseDTO> lista = new ArrayList<>();
        for (Medico m : medicoRepository.findByHospitalId(hospitalId)) {
            lista.add(mapearADTO(m));
        }
        return lista;   // una lista vacía es válida
    }

    @Override
    @Transactional
    public void eliminar(Long id) {
        Medico medico = buscarMedico(id);
        if (consultaRepository.countByMedicoId(id) > 0) {
            throw new ConflictoException("El médico tiene consultas registradas");
        }
        medicoRepository.delete(medico);
    }

    private Hospital buscarHospital(Long id) {
        return hospitalRepository.findById(id)
                .orElseThrow(() -> new NoEncontradoException("No existe el hospital " + id));
    }

    private Medico buscarMedico(Long id) {
        return medicoRepository.findById(id)
                .orElseThrow(() -> new NoEncontradoException("No existe el médico " + id));
    }

    // Entity -> DTO (el hospital es LAZY: por eso se hace dentro de la transacción)
    private MedicoResponseDTO mapearADTO(Medico m) {
        return new MedicoResponseDTO(m.getId(), m.getNombre(), m.getEspecialidad(), m.getHospital().getNombre());
    }
}

package com.hospital.examen.exception;

import java.util.Map;
import java.util.stream.Collectors;

import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.http.converter.HttpMessageNotReadableException;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;

@RestControllerAdvice
public class ManejadorErrores {

    @ExceptionHandler(NoEncontradoException.class)
    public ResponseEntity<Map<String, Object>> noEncontrado(NoEncontradoException e) {
        return respuesta(HttpStatus.NOT_FOUND, e.getMessage());
    }

    @ExceptionHandler(ConflictoException.class)
    public ResponseEntity<Map<String, Object>> conflicto(ConflictoException e) {
        return respuesta(HttpStatus.CONFLICT, e.getMessage());
    }

    // Errores de @Valid: se juntan los mensajes de todos los campos incorrectos
    @ExceptionHandler(MethodArgumentNotValidException.class)
    public ResponseEntity<Map<String, Object>> validacion(MethodArgumentNotValidException e) {
        String mensaje = e.getBindingResult().getFieldErrors().stream()
                .map(f -> f.getField() + ": " + f.getDefaultMessage())
                .collect(Collectors.joining("; "));
        return respuesta(HttpStatus.BAD_REQUEST, mensaje);
    }

    // JSON mal escrito, sin cuerpo o con un valor de enum que no existe
    @ExceptionHandler(HttpMessageNotReadableException.class)
    public ResponseEntity<Map<String, Object>> cuerpoIlegible(HttpMessageNotReadableException e) {
        return respuesta(HttpStatus.BAD_REQUEST, "El cuerpo de la petición no es válido");
    }

    private ResponseEntity<Map<String, Object>> respuesta(HttpStatus estado, String mensaje) {
        return ResponseEntity.status(estado).body(Map.of("mensaje", mensaje, "codigo", estado.value()));
    }
}

Parte 5: preguntas de comprensión

22. LazyInitializationException. Ocurre al leer una relación LAZY (aquí, las consultas del paciente, o el médico de cada consulta) cuando la sesión de Hibernate ya se ha cerrado: por ejemplo, si el servicio devuelve la entidad y es el controlador quien intenta recorrerla. Se evita convirtiendo a DTO dentro del servicio y anotándolo con @Transactional (readOnly = true si solo lee), para que la sesión siga abierta mientras se recorren las relaciones.

23. El bucle infinito. La causa es que la relación es bidireccional: Hospital tiene la lista de médicos y cada Medico apunta a su Hospital. Al convertir a JSON, Jackson recorre hospital → médicos → hospital → médicos… sin fin. Dos soluciones:

  • Usar DTO en vez de entidades (la buena): el DTO de salida no tiene referencia de vuelta, o solo un resumen (nombreHospital).
  • Anotaciones de Jackson sobre las entidades: @JsonIgnore en el lado que no queremos serializar, o la pareja @JsonManagedReference / @JsonBackReference. Funciona, pero ata la entidad al formato JSON, por eso no es lo recomendable.

24. Borrar un hospital con médicos. El 500 viene de la base de datos (violación de clave foránea: los médicos siguen apuntando al hospital). Debería devolver 409 Conflict. Falta la comprobación previa en el servicio, igual que con las consultas de un médico:
